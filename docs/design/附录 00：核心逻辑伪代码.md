### 1. GM 审批移动（立即执行 vs 预约执行）

**场景**：GM 在审批弹窗中处理玩家的预约移动申请。

```javascript
// 文件: services/moveService.js (伪代码)

function handleGMApproveMove(moveId, gmUserId, arrivalStoryTime) {
  // arrivalStoryTime: 剧情到达时间，格式 { day, hour, minute }；null 表示立即执行
  const move = db.scheduled_moves.findById(moveId);
  
  // 1. 校验
  if (!move || move.status !== 'pending') {
    return { error: 'INVALID_MOVE', message: '预约不存在或已处理' };
  }
  
  const targetScene = db.scenes.findById(move.to_scene_id);
  if (!targetScene || targetScene.status !== 'active') {
    // 目标场景不可用 → 自动取消
    db.scheduled_moves.update(moveId, { status: 'cancelled', cancel_reason: 'scene_closed' });
    notifyPlayer(move.character_id, '您的移动申请已取消：目标场景已关闭');
    return { error: 'SCENE_NOT_ACTIVE' };
  }
  
  // 2. 分支：立即执行 vs 预约执行
  if (arrivalStoryTime === null) {
    // --- 立即执行分支 ---
    db.transaction(() => {
      // 2.1 更新角色位置
      const fromSceneId = getCurrentSceneId(move.character_id, move.campaign_id);
      updateCharacterPosition(move.character_id, move.to_scene_id);
      
      // 2.2 记录位置历史（剧情到达时间留空）
      insertPositionHistory({
        character_id: move.character_id,
        scene_id: move.to_scene_id,
        story_time_entered: null,
        move_type: 'approved'
      });
      
      // 2.3 更新预约状态
      db.scheduled_moves.update(moveId, { status: 'executed' });
      
      // 2.4 发送场景事件消息（受时序裁剪）
      if (fromSceneId) {
        sendSceneEvent(fromSceneId, `${move.character_name} 离开当前场景`);
      }
      sendSceneEvent(move.to_scene_id, `${move.character_name} 进入当前场景`);
    });
    
    // 2.5 通知双方（通过 Socket）
    broadcastToCampaign(move.campaign_id, 'position_changed', {
      character_id: move.character_id,
      from_scene_id: fromSceneId,
      to_scene_id: move.to_scene_id,
      move_type: 'approved'
    });
    
    return { success: true, executed_immediately: true };
    
  } else {
    // --- 预约执行分支 ---
    db.transaction(() => {
      db.scheduled_moves.update(moveId, {
        status: 'approved',
        execute_at_story: arrivalStoryTime   // 存储剧情时间（JSON）
      });
    });
    
    // 通知玩家
    notifyPlayer(move.character_id, 
      `您的移动申请已批准，将于剧情时间 ${formatStoryTime(arrivalStoryTime)} 自动执行`);
    
    // 在 GM 助理台的“即将自动执行”列表中展示
    addToPendingTimeline(move.campaign_id, {
      move_id: moveId,
      character_name: move.character_name,
      to_scene_name: targetScene.name,
      execute_at: arrivalStoryTime
    });
    
    return { success: true, scheduled: true, execute_at: arrivalStoryTime };
  }
}
```

### 2. 时间推进与预约移动自动调度

**场景**：GM 宣布时间推进（固定步进或自定义时间），系统自动触发所有已到期的预约移动。

```javascript
// 文件: services/timeService.js (伪代码)

function advanceGlobalTime(campaignId, gmUserId, newStoryTime) {
  // newStoryTime: { day, hour, minute }，必然 ≥ 当前全局时间（前端保证）
  const campaign = db.campaigns.findById(campaignId);
  const currentTime = campaign.global_story_time;
  
  // 1. 如果推进幅度 > 2小时，前端应已弹窗二次确认，后端无需重复判断，直接继续
  
  // 2. 查询所有状态为 'approved' 且 execute_at_story ≤ newStoryTime 的预约移动
  const approvedMoves = db.scheduled_moves.find({
    campaign_id: campaignId,
    status: 'approved',
    execute_at_story: { $lte: newStoryTime }
  });
  
  // 3. 按 execute_at_story 升序排序（保证因果顺序）
  approvedMoves.sort((a, b) => compareStoryTime(a.execute_at_story, b.execute_at_story));
  
  const executedMoves = [];
  
  for (const move of approvedMoves) {
    // 3.1 再次检查目标场景是否仍 active（避免期间被归档）
    const targetScene = db.scenes.findById(move.to_scene_id);
    if (!targetScene || targetScene.status !== 'active') {
      db.scheduled_moves.update(move.id, { status: 'cancelled', cancel_reason: 'scene_closed' });
      notifyPlayer(move.character_id, `前往 ${targetScene?.name || '未知场景'} 的预约已取消（目标场景已关闭）`);
      continue;
    }
    
    // 3.2 检查角色是否仍在团中且未死亡
    const charInstance = db.character_instances.findOne({
      character_id: move.character_id,
      campaign_id: campaignId,
      status: 'active'
    });
    if (!charInstance || charInstance.status === 'dead') {
      db.scheduled_moves.update(move.id, { status: 'cancelled', cancel_reason: 'character_inactive' });
      continue; // 静默取消，不发通知
    }
    
    // 3.3 执行移动（复用立即执行逻辑，但不通过 GM 审批）
    const fromSceneId = charInstance.current_spatial_scene_id;
    updateCharacterPosition(move.character_id, move.to_scene_id);
    insertPositionHistory({
      character_id: move.character_id,
      scene_id: move.to_scene_id,
      story_time_entered: move.execute_at_story,   // 预约的到达时间
      move_type: 'scheduled'
    });
    db.scheduled_moves.update(move.id, { status: 'executed', executed_at_story: newStoryTime });
    
    // 3.4 发送场景事件消息（仅发送给目标场景当前角色可见的可见范围）
    if (fromSceneId) {
      sendSceneEvent(fromSceneId, `${move.character_name} 离开当前场景`);
    }
    sendSceneEvent(move.to_scene_id, `${move.character_name} 进入当前场景（预约到达）`);
    
    executedMoves.push({
      character_name: move.character_name,
      from_scene_name: getSceneName(fromSceneId),
      to_scene_name: targetScene.name,
      arrive_at: move.execute_at_story
    });
  }
  
  // 4. 更新团的全局剧情时间
  db.campaigns.update(campaignId, { global_story_time: newStoryTime });
  
  // 5. 在聊天流插入 time_tag 消息（豁免时序裁剪）
  insertTimeTagMessage(campaignId, newStoryTime);
  
  // 6. 如果执行了任何移动，向 GM 助理台推送摘要
  if (executedMoves.length > 0) {
    sendToGM(campaignId, 'auto_moves_executed', {
      time_label: formatStoryTime(newStoryTime),
      moves: executedMoves
    });
  }
  
  // 7. 广播时间推进事件给所有在线成员（不含移动细节，只更新界面时间显示）
  broadcastToCampaign(campaignId, 'time_advanced', {
    old_time: currentTime,
    new_time: newStoryTime,
    triggered_move_count: executedMoves.length
  });
  
  return { success: true, executed_moves: executedMoves };
}
```

### 3. 成团流程（从招募帖到创建房间）

**场景**：GM 在招募帖点击“成团”，系统创建房间并关联成员。

```javascript
// 文件: services/recruitmentService.js (伪代码)

function finalizeRecruitment(postId, gmUserId, selectedPlayerIds, moduleChoice) {
  // moduleChoice: { type: 'selected' | 'other' | 'blank', module_id?: string }
  
  const post = db.recruitment_posts.findById(postId);
  
  // 1. 校验
  if (!post || post.status !== 'open') {
    return { error: 'RECRUITMENT_NOT_OPEN' };
  }
  if (post.gm_user_id !== gmUserId) {
    return { error: 'PERMISSION_DENIED' };
  }
  
  // 2. 获取已确认入团的玩家（状态 = 'confirmed'）
  const confirmedApps = db.applications.find({
    recruitment_id: postId,
    status: 'confirmed'
  });
  const confirmedPlayerIds = confirmedApps.map(a => a.user_id);
  
  // 若 GM 手动勾选，则取交集（不能添加未确认的玩家）
  const finalPlayerIds = selectedPlayerIds.filter(id => confirmedPlayerIds.includes(id));
  if (finalPlayerIds.length === 0) {
    return { error: 'NO_CONFIRMED_PLAYERS' };
  }
  
  // 3. 确定模组
  let finalModuleId = null;
  if (moduleChoice.type === 'selected') {
    if (post.module_id) {
      finalModuleId = post.module_id;   // 发帖时已选定的模组
    } else {
      return { error: 'NO_MODULE_SELECTED' };
    }
  } else if (moduleChoice.type === 'other') {
    finalModuleId = moduleChoice.module_id;
    // 校验 GM 是否拥有该模组
    if (!userOwnsModule(gmUserId, finalModuleId)) {
      return { error: 'MODULE_NOT_OWNED' };
    }
  } else if (moduleChoice.type === 'blank') {
    finalModuleId = null;   // 创建空白团
  }
  
  // 4. 创建房间（Campaign）
  const campaignId = generateUUID();
  const roomCode = generateRoomCode();
  
  db.transaction(() => {
    db.campaigns.insert({
      id: campaignId,
      room_code: roomCode,
      name: post.title,   // 默认使用招募帖标题，GM 可后续修改
      ruleset_id: post.ruleset_id,
      module_id: finalModuleId,
      gm_user_id: gmUserId,
      status: 'preparing',
      global_story_time: { day: 1, hour: 8, minute: 0 }, // 默认初始时间
      created_at: now()
    });
    
    // 5. 将 GM 加入房间（作为 GM 角色）
    addUserToCampaign(campaignId, gmUserId, role='gm');
    
    // 6. 将选中的玩家加入房间，并绑定他们的角色卡（角色卡 ID 从 application 中获取）
    for (const userId of finalPlayerIds) {
      const app = confirmedApps.find(a => a.user_id === userId);
      if (!app || !app.character_id) {
        // 若玩家申请时未绑定角色卡，需弹窗让玩家选择（在成团前应已处理）
        continue;
      }
      addUserToCampaign(campaignId, userId, role='player', character_id=app.character_id);
    }
    
    // 7. 更新招募帖状态 + 关联房间
    db.recruitment_posts.update(postId, {
      status: 'grouped',
      campaign_id: campaignId
    });
    
    // 8. 创建默认公共场
    createDefaultLobby(campaignId);
    
    // 9. 如果绑定了模组，自动导入模组中的场景、NPC、线索等
    if (finalModuleId) {
      importModuleContent(campaignId, finalModuleId);
    } else {
      // 空白团：至少创建一个默认剧情场，让 GM 有地方可去
      createDefaultSpatialScene(campaignId, '起始地点');
    }
  });
  
  // 10. 发送通知
  for (const userId of finalPlayerIds) {
    notifyUser(userId, 'campaign_ready', {
      campaign_id: campaignId,
      campaign_name: post.title,
      room_code: roomCode,
      message: `你参与的招募「${post.title}」已成团，点击进入房间`
    });
  }
  
  return { success: true, campaign_id: campaignId, room_code: roomCode };
}
```

### 4. 技能成长跨团同步（Template → Instance）

**场景**：玩家在某个团中技能成长后（`en` 指令成功），如何同步到角色卡模板（Template）以及其他活跃团中的实例（Instance）。

```javascript
// 文件: services/skillGrowthService.js (伪代码)

function applySkillGrowth(characterId, skillId, growthValue, sourceCampaignId) {
  // growthValue: 通常为 1d10 的结果，直接加到技能值上（不超过 max）
  
  // 1. 更新角色卡模板（Template）
  const template = db.character_sheets.findById(characterId);
  const oldSkillValue = template.skills[skillId] || 0;
  const newSkillValue = Math.min(oldSkillValue + growthValue, 99); // 设最大 99
  db.character_sheets.update(characterId, {
    skills: { ...template.skills, [skillId]: newSkillValue }
  });
  
  // 2. 查找该角色卡当前所有活跃的团实例（status='active'）
  const activeInstances = db.character_instances.find({
    character_id: characterId,
    status: 'active'
  });
  
  const syncResults = [];
  
  for (const instance of activeInstances) {
    const instanceCampaignId = instance.campaign_id;
    const instanceSkillValue = instance.skills?.[skillId] || 0;
    
    // 2.1 如果当前实例的技能值已经 >= 新模板值，跳过
    if (instanceSkillValue >= newSkillValue) {
      syncResults.push({ campaign_id: instanceCampaignId, synced: false, reason: 'already_higher' });
      continue;
    }
    
    // 2.2 如果成长差距 ≥ 5 点，强制同步（不弹窗）
    if (newSkillValue - instanceSkillValue >= 5) {
      db.character_instances.update(instance.id, {
        skills: { ...instance.skills, [skillId]: newSkillValue }
      });
      // 在聊天流插入系统消息（该团可见）
      insertSystemMessage(instanceCampaignId, 
        `[系统] 角色「${template.name}」的技能「${skillId}」已自动同步至 ${newSkillValue}`);
      syncResults.push({ campaign_id: instanceCampaignId, synced: true, reason: 'force_sync' });
      
    } else {
      // 2.3 差距 < 5，标记为“待同步”，下次进入房间时提示 GM/玩家
      markPendingSync(instance.id, skillId, newSkillValue);
      syncResults.push({ campaign_id: instanceCampaignId, synced: false, reason: 'pending_confirmation' });
    }
  }
  
  // 3. 如果当前触发成长的团是活跃实例之一，直接将实例技能值更新（已在上面覆盖或标记）
  // 注意：如果来源团的实例因为差距<5而未被强制同步，则本次成长在该团内也不会立即生效，
  // 玩家会看到提示“技能已成长到 X，是否同步？”（由前端检测 pending_sync 标记后弹窗）
  
  return {
    success: true,
    template_new_value: newSkillValue,
    sync_results: syncResults
  };
}

// 辅助：玩家进入房间时检测待同步标记
function onPlayerJoinCampaign(characterId, campaignId) {
  const pendingSyncs = db.pending_skill_syncs.find({
    character_id: characterId,
    campaign_id: campaignId,
    resolved: false
  });
  
  if (pendingSyncs.length > 0) {
    // 前端收到此事件后，在顶部横幅或助理台显示“技能可同步”提示
    notifyClient(campaignId, characterId, 'skill_sync_available', {
      skills: pendingSyncs.map(s => ({ skill_id: s.skill_id, new_value: s.new_value }))
    });
  }
}
```

### 5. 候补转正自动邀请

**场景**：当有 `confirmed` 玩家退出或超时未确认时，系统从 `waiting` 队列中自动邀请下一位。

```javascript
// 文件: services/waitlistService.js (伪代码)

function onSlotReleased(recruitmentId) {
  // 1. 获取该招募帖当前信息
  const post = db.recruitment_posts.findById(recruitmentId);
  if (post.status !== 'open' && post.status !== 'full') return;
  
  // 2. 查询当前 confirmed 人数
  const confirmedCount = db.applications.count({
    recruitment_id: recruitmentId,
    status: 'confirmed'
  });
  const maxSlots = post.player_count_max;
  const availableSlots = maxSlots - confirmedCount;
  
  if (availableSlots <= 0) return;
  
  // 3. 从 waiting 队列中取出前 N 个（按申请时间升序）
  const waitingApps = db.applications.find({
    recruitment_id: recruitmentId,
    status: 'waiting'
  }, { sort: { created_at: 1 } });
  
  const toInvite = waitingApps.slice(0, availableSlots);
  
  for (const app of toInvite) {
    db.transaction(() => {
      // 3.1 将状态从 waiting 改为 invited，并设定邀请有效期（24h）
      db.applications.update(app.id, {
        status: 'invited',
        invited_expire_at: now() + 24 * 3600 * 1000
      });
      
      // 3.2 发送通知
      notifyUser(app.user_id, 'waitlist_promoted', {
        recruitment_id: recruitmentId,
        recruitment_title: post.title,
        expire_at: app.invited_expire_at
      });
    });
  }
  
  // 4. 如果 new confirmed + waiting 未满，招募帖状态保持 open；若释放后仍有空位但无 waiting，则也保持 open
  const remainingWaiting = db.applications.count({
    recruitment_id: recruitmentId,
    status: 'waiting'
  });
  
  if (confirmedCount + remainingWaiting < maxSlots) {
    db.recruitment_posts.update(recruitmentId, { status: 'open' });
  } else if (confirmedCount + remainingWaiting >= maxSlots) {
    db.recruitment_posts.update(recruitmentId, { status: 'full' });
  }
}

// 2. 邀请超时（定时任务扫描 invited_expire_at < now()）
// 3. GM 手动将某个 confirmed 玩家移出团（在成团前）
```
